Consider the following function.
#+begin_src ocaml
  let output (t : task) (x : int) =
    match t with
    | Apply (f, y) -> f x (y + 1)
    | Flip (l) -> List.map (fun e -> not e) l
    | Filter (f) -> List.filter (f.pred x) f.elems
    | Reverse (l) -> List.fold_left (fun b a -> a :: b) [] l
#+end_src

Determine the *smallest* type ~task~ such that ~output~ is
well-typed.
