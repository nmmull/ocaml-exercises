Consider the following function.
#+begin_src ocaml
  let output (t : task) (x : int) =
    match t with
    | Compute (f) -> f x
    | Double (l) -> List.map (fun e -> 2 * e) l
    | Filter (p, l) -> List.filter (p x) l
    | Append (l) -> List.fold_right (fun a b -> a :: b) l.second l.first
#+end_src

Determine the *smallest* type ~task~ such that ~output~ is
well-typed.
