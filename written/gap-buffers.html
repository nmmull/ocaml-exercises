Gap Buffers

A gap buffer is a data structure common in the implementation of text editors. It's like a list, but with an interface for moving and accessing a cursor location. Below is a toy OCaml implementation of a gap buffer. 

let empty : 'a gbuf = { lft = []; rgt = End }

let get_cur (c : 'a cur) : 'a option =
  match c with
  | End -> None
  | Cur (x, _) -> Some x

let get (g : 'a gbuf) : 'a option = get_cur g.rgt

let insert (x : 'a) (g : 'a gbuf) : 'a gbuf = { lft = x :: g.lft; rgt = g.rgt }

let move_left (g : 'a gbuf) : 'a gbuf =
  match g.lft with
  | [] -> g
  | x :: xs ->
    let r =
      match g.rgt with
      | End -> []
      | Cur (y, ys) -> y :: ys
    in
    { lft = xs; rgt = Cur (x, r) }

let ex = get (move_left (move_left (insert 3 (insert 2 (insert 1 empty)))))
