Implement the function
#+begin_src ocaml
  val curry : (('a * 'b) -> 'c) -> 'a -> 'b -> 'c
#+end_src
which converts a "multi-argument" function (i.e., a function which
takes a tuple as an argument) into nested single-argument functions.

** Examples

#+begin_src ocaml
  let f (x, y) = x + y
  let g = curry f
  let _ = assert (g 2 3 = 5)
#+end_src
