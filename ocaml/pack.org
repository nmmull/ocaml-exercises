Implement the function
#+begin_src ocaml
  val pack : 'a list -> 'a list list
#+end_src
so that ~pack l~ packs together consecutive elements of ~l~ that are
equal into a sublist.  For instance, with the input
~[1;1;2;3;3;2;4;4;4]~, the output should be
~[[1;1];[2];[3;3];[2];[4;4;4]]~.  Note that the second occurrence of
~2~ is not packed with the first occurrence of ~2~, since only
*consecutive* elements that are equal are packed together.

** Examples

#+begin_src ocaml
  let _ = assert (pack [] = [])
  let _ = assert (pack [1] = [[1]])
  let _ = assert (pack [1;1;2;3;3;2;4;4;4] = [[1;1];[2];[3;3];[2];[4;4;4]])
  let _ = assert (pack [1;2;3] = [[1];[2];[3]])
#+end_src
