Implement the function
#+begin_src ocaml
  val min_sos : int -> int
#+end_src
so that ~min_sos n~ is the minimum number of squares required to sum
up to ~n~.

** Examples

#+begin_src ocaml
  let _ = assert (min_sos 3 = 3)    (* 3 = 1^2 + 1^2 + 1^2 *)
  let _ = assert (min_sos 4 = 1)    (* 4 = 2^2 *)
  let _ = assert (min_sos 5 = 2)    (* 5 = 2^2 + 1^2 *)
#+end_src
