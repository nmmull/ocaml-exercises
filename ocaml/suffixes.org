Implement the function
#+begin_src ocaml
  val suffixes : 'a list -> 'a list list
#+end_src
so that ~suffixes l~ is the list of all suffixes of ~l~ in decreasing
order of length (~suf~ is a suffix of a list ~l~ if there is a list
~pre~ such that ~pre @ suf = l~).

** Examples

#+begin_src ocaml
  let _ = assert (suffixes [1;2;3] = [[1;2;3]; [2;3]; [3]; []])
#+end_src
