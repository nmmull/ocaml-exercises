Implement the function

#+begin_src ocaml
  val fun_max : (int -> int) list -> int -> int
#+end_src
so that ~fun_max fs x~ is given by
\[
\mathsf{funMax}(\texttt{fs}, x) = \max\left(\max_{f \in \mathtt{fs}}\{f(x)\}, 0\right).
\]

** Examples

#+begin_src ocaml
  let _ = assert (fun_max [(+) 1; fun x -> x * x] 1 = 2)
  let _ = assert (fun_max [(+) 1; fun x -> x * x] (-2) = 4)
#+end_src
