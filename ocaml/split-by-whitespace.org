Implement the function
#+begin_src ocaml
  val split_on_ws : string -> string list
#+end_src
so that ~split_on_ws s~ is the list of contiguous non-whitespace
substrings of ~s~, in the order in which they appear.

For this version of the problem, you may work with strings as lists of
characters, using the functions ~explode~ and ~implode~ to convert
between strings and character lists.

#+begin_src ocaml
  let explode (s : string) : char list =
    List.init (String.length s) (String.get s)

  let implode (l : char list) : string =
    String.init (List.length l) (List.nth l)
#+end_src

** Examples

#+begin_src ocaml
  let _ = assert (split_on_ws "1 23 + -94  *" = ["1"; "23"; "+"; "-94"; "*"])
  let _ = assert (split_on_ws "  foo    bar  \n\n   baz   " = ["foo"; "bar"; "baz"])
#+end_src
