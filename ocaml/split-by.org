Without using any functions from the standard library (except for
basic constructors like ~(::)~ and ~[]~), implement the function
#+begin_src ocaml
  val split_by : ('a -> bool) -> 'a list -> 'a list * 'a list
#+end_src
so that ~split_by f l~ is a pair of lists ~(lt, lf)~ where ~lt~
consists of those elements of ~l~ (in order, with repetitions) which
satisfy the predicate ~f~, and ~lf~ consists of those elements which
do not satisfy the predicate ~f~.  *Your solution must be tail
recursive.*

** Examples

#+begin_src ocaml
  let _ = assert (split_by (fun x -> x > 3) [0;5;4;2;1;4] = ([5;4;4], [0;2;1]))
  let _ = assert (split_by (fun x -> x = 2 || x = 4) [1;2;3;4;5;6]
                  = ([2;4], [1;3;5;6]))
#+end_src
