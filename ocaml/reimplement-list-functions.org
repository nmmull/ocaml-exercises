Without using any functions from the standard library, implement
~map~, ~filter~, ~fold_right~, and ~fold_left~ for lists.

#+begin_src ocaml
  val map : ('a -> 'b) -> 'a list -> 'b list
  val filter : ('a -> bool) -> 'a list -> 'a list
  val fold_right : ('a -> 'b -> 'b) -> 'a list -> 'b -> 'b
  val fold_left : ('b -> 'a -> 'b) -> 'b -> 'a list -> 'b
#+end_src

For an added challenge, make them tail recursive.

** Examples

#+begin_src ocaml
  let _ = assert (map (fun x -> x + 1) [1;2;3] = [2;3;4])
  let _ = assert (filter (fun x -> x mod 2 = 0) [1;2;3;4;5;6] = [2;4;6])
  let _ = assert (fold_right (fun x acc -> x :: acc) [1;2;3] [] = [1;2;3])
  let _ = assert (fold_right (-) [1;2;3] 0 = 1 - (2 - (3 - 0)))
  let _ = assert (fold_left (fun acc x -> acc - x) 0 [1;2;3] = ((0 - 1) - 2) - 3)
#+end_src
