A *rose tree* is represented by an ADT with a single constructor, a
~Node~ with a value and a list of children.

#+begin_src ocaml
  type 'a rtree = Node of 'a * 'a rtree list
#+end_src

Note that it is possible for a ~Node~ to have no children.  For
example,

#+begin_src ocaml
  let example : int rtree =
    Node
      ( 1
      , [ Node (2, [])
        ; Node (3, [Node (4, []); Node (5, [])])
        ; Node (6, [])
        ]
      )
#+end_src

** Task 1

Implement the function
#+begin_src ocaml
  val map_rtree : ('a -> 'b) -> 'a rtree -> 'b rtree
#+end_src
so that ~map_rtree f t~ is the result of applying ~f~ to every value
of ~t~, maintaining the structure of ~t~.

#+begin_src ocaml
  let _ = assert (map_rtree (fun x -> x * x) example =
      Node (1, [Node (4, []); Node (9, [Node (16, []); Node (25, [])]); Node (36, [])]))
#+end_src

** Task 2

Implement the function
#+begin_src ocaml
  val filter_rtree : ('a -> bool) -> 'a rtree -> 'a rtree option
#+end_src
so that ~filter_rtree f t~ is the result of filtering out those
values ~v~ of ~t~ for which ~f v = false~.  In particular, for a
subtree ~Node (v, ts)~, if ~f v = false~ then /the whole subtree/
should be removed.

#+begin_src ocaml
  let _ = assert (filter_rtree (fun x -> x <> 3) example =
      Some (Node (1, [Node (2, []); Node (6, [])])))
  let _ = assert (filter_rtree (fun x -> x <> 1) example = None)
#+end_src
