Split Tree

type 'a ntree = Node of 'a * 'a ntree list

Implement the function 

val split_tree : ('a * 'b) ntree -> 'a ntree * 'b ntree

so that split_tree t constructs two trees of the same structure as t, one with the left element of the tuple in each node of t, and one with the right element of the tuple in each node.

Examples

let t = Node ((1, "a"), [Node ((2, "b"), []); Node ((3, "c"), [])])
let _ = assert (split_tree t =
    ( Node (1, [Node (2, []); Node (3, [])])
    , Node ("a", [Node ("b", []); Node ("c", [])])
    ))
let _ = assert (split_tree (Node ((1, "a"), [])) = (Node (1, []), Node ("a", [])))
