#+begin_src ocaml
  type dir = N | S | E | W
#+end_src

Implement the function
#+begin_src ocaml
  val dist : dir list -> float
#+end_src
so that ~dist dirs~, given a list of directions ~dirs~, computes the
$\ell_2$ distance (i.e., "as the crow flies") in meters between the
starting point and ending point of the path given by ~dirs~.  Note
that it doesn't matter where you start, the distance between the
starting and ending points will always be the same.

** Examples

#+begin_src ocaml
  let is_close f1 f2 = Float.abs (f1 -.f2) < 10. ** (-15.)
  let _ = assert (is_close (dist [North; North; South; East]) (sqrt 2.))
#+end_src
