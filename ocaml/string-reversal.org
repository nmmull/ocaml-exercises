Implement the function
#+begin_src ocaml
  val rev_string : string -> string
#+end_src
so that ~rev_string s~ is the string with the same characters as ~s~
but in reverse order.  Do this without converting the string into a
list of characters.

** Examples

#+begin_src ocaml
  let _ = assert (rev_string "testing" = "gnitset")
  let _ = assert (rev_string "12345" = "54321")
  let _ = assert (rev_string "noon" = "noon")
#+end_src
