Implement the function
#+begin_src ocaml
  val drop_leading : int -> int list -> int list
#+end_src
so that ~drop_leading k l~ is the result of removing all instances
of ~k~ from the beginning of the list ~l~.

** Examples

#+begin_src ocaml
  let _ = assert (drop_leading 1 [1;1;2;1;3] = [2;1;3])
  let _ = assert (drop_leading 5 [1;2;3] = [1;2;3])
  let _ = assert (drop_leading 2 [2;2;2] = [])
#+end_src
